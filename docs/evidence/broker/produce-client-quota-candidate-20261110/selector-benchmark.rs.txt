    #[test]
    #[ignore = "diagnostic measurement, not a benchmark qualification"]
    fn quota_selector_cost_samples() {
        use std::hint::black_box;
        let now = Instant::now();
        let quota = ProduceQuota::new(1_000, 100, 10_000).unwrap();
        let limits = crate::protocol::Limits::default();
        let mut modes = Vec::new();
        for count in [0usize, 1, 64, 256] {
            let target = if count == 0 { 0 } else { count / 2 };
            let request = header(9, Some(&format!("id-{target:03}")));
            let policy = if count == 0 {
                QuotaPolicy::Shared(quota)
            } else {
                let names = (0..count).map(|index| (format!("id-{index:03}"), quota)).collect();
                QuotaPolicy::Clients(ClientProduceQuotas::new(quota, names).unwrap())
            };
            modes.push((count, QuotaMeters::new(policy, now), request));
        }
        const ITERATIONS: usize = 1_000_000;
        for round in 0..8 {
            for step in 0..modes.len() {
                let index = if round % 2 == 0 { step } else { modes.len() - 1 - step };
                let (count, meter, request) = &modes[index];
                let started = Instant::now();
                let mut hits = 0usize;
                for _ in 0..ITERATIONS {
                    let selected = black_box(meter).select(black_box(request), black_box(limits));
                    hits += usize::from(black_box(selected).is_some());
                }
                let elapsed = started.elapsed().as_nanos();
                assert_eq!(hits, if *count == 0 { 0 } else { ITERATIONS });
                println!("sample mode={count} round={round} iterations={ITERATIONS} elapsed_ns={elapsed}");
            }
        }
    }
