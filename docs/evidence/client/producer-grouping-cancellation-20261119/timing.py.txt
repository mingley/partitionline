"""Run paired diagnostics using two independently built release test binaries."""

import hashlib
import os
import platform
import re
import statistics
import subprocess
import sys
from pathlib import Path


SHAPES = {
    "single-1": 100_000,
    "single-64": 20_000,
    "single-8193": 128,
    "partitioned-64": 20_000,
    "mixed-512": 2_000,
    "mixed-scatter-8193": 128,
    "mixed-assembly-2051": 512,
}
SAMPLE = re.compile(r"GROUPING_TIMING case=([a-z0-9-]+) iterations=(\d+) ns=(\d+)")


def read_binary(path_file):
    path = Path(path_file).read_text().strip()
    binary_path = Path(path).resolve(strict=True)
    digest = hashlib.file_digest(binary_path.open("rb"), "sha256").hexdigest()
    return binary_path, digest


def measure(binary):
    result = subprocess.run(
        [binary, "--exact", "--ignored", "--nocapture", "--test-threads=1",
         "producer::tests::grouping_timing_diagnostic"],
        check=True,
        capture_output=True,
        text=True,
        timeout=90,
    )
    if "test result: ok. 1 passed; 0 failed; 0 ignored" not in result.stdout:
        raise ValueError(result.stdout)
    measured = {}
    for shape, count, elapsed in SAMPLE.findall(result.stdout):
        if shape in measured or int(count) != SHAPES.get(shape) or int(elapsed) <= 0:
            raise ValueError(f"invalid sample for {shape}")
        measured[shape] = int(elapsed)
    if set(measured) != set(SHAPES):
        raise ValueError(f"incomplete samples: {result.stdout}")
    return measured


def main():
    baseline, baseline_hash = read_binary(sys.argv[1])
    candidate, candidate_hash = read_binary(sys.argv[2])
    if baseline == candidate or baseline_hash == candidate_hash:
        raise ValueError("paired builds must yield distinct binaries")
    if platform.machine() != "x86_64":
        raise ValueError("this diagnostic is scoped to x86_64")
    cpu_set = os.sched_getaffinity(0)
    cpu = min(cpu_set)
    os.sched_setaffinity(0, {cpu})
    print(f"GROUPING_HOST arch={platform.machine()} cpu={cpu} available={len(cpu_set)}", flush=True)
    print(f"GROUPING_BINARY baseline={baseline_hash} candidate={candidate_hash}", flush=True)

    binaries = [baseline, candidate]
    pair_ratios = {shape: [] for shape in SHAPES}
    for pair in range(1, 9):
        order = [0, 1] if pair % 2 else [1, 0]
        rows = {}
        for arm in order:
            for shape, elapsed in measure(binaries[arm]).items():
                rows[arm, shape] = elapsed
                print(f"GROUPING_SAMPLE pair={pair} arm={['baseline', 'candidate'][arm]} "
                      f"case={shape} iterations={SHAPES[shape]} ns={elapsed}", flush=True)
        for shape, ratios in pair_ratios.items():
            ratios.append(rows[1, shape] / rows[0, shape])

    for shape, ratios in pair_ratios.items():
        print(f"GROUPING_RATIO case={shape} median={statistics.median(ratios):.4f} "
              f"min={min(ratios):.4f} max={max(ratios):.4f}", flush=True)


if __name__ == "__main__":
    main()
