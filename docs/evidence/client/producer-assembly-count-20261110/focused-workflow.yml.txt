    branches: [qualification/producer-assembly-clean-20261110]

    timeout-minutes: 25
    steps:

      - name: Default assembly controls

      - name: Default existing assembly behavior

        if: always()

        if: always()
        run: cargo test --locked --lib producer::tests::grouping_ -- --nocapture
      - name: All-feature grouping behavior
        if: always()

      - name: Format
        if: always()
        run: cargo fmt --all -- --check
      - name: Strict Clippy
        if: always()
